function toast(message,type='ok'){
  const t=$('#toast'); t.textContent=message; t.className=`toast show ${type==='error'?'error':''}`;
  clearTimeout(window.__toastTimer); window.__toastTimer=setTimeout(()=>t.className='toast',3500);
}

function openModal(html){ $('#modalContent').innerHTML=html; $('#modal').classList.remove('hidden'); $('#modal').setAttribute('aria-hidden','false'); }

function showAuth(){ $('#authView').classList.remove('hidden'); $('#appView').classList.add('hidden'); }

async function init(){
  bindGlobalEvents();
  supabase.auth.onAuthStateChange(async(event,s)=>{
    session=s;
    if(event==='PASSWORD_RECOVERY'){
      showAuth(); $('#loginForm').classList.add('hidden'); $('#recoveryBtn').classList.add('hidden'); $('#passwordForm').classList.remove('hidden');
      return;
    }
    if(s) await enterApp(); else showAuth();
  });
  const {data:{session:s}}=await supabase.auth.getSession(); session=s;
  if(s) await enterApp(); else showAuth();
}

function bindGlobalEvents(){
  $('#loginForm').addEventListener('submit',async e=>{
    e.preventDefault(); $('#loginError').textContent='';
    const {error}=await supabase.auth.signInWithPassword({email:val('#loginEmail'),password:val('#loginPassword')});
    if(error) $('#loginError').textContent=error.message;
  });
  $('#recoveryBtn').addEventListener('click',async()=>{
    const email=val('#loginEmail');
    if(!email){ $('#loginError').textContent='Inserisci prima l’email.'; return; }
    const {error}=await supabase.auth.resetPasswordForEmail(email,{redirectTo:location.origin+location.pathname});
    $('#loginError').textContent=error?error.message:'Ti abbiamo inviato il link per impostare o recuperare la password.';
  });
  $('#passwordForm').addEventListener('submit',async e=>{
    e.preventDefault(); const password=val('#newPassword'); const {error}=await supabase.auth.updateUser({password});
    if(error){$('#loginError').textContent=error.message;return;} toast('Password aggiornata.'); $('#passwordForm').classList.add('hidden'); await enterApp();
  });
  $('#logoutBtn').addEventListener('click',()=>supabase.auth.signOut());
  $('#mobileNavBtn').addEventListener('click',()=>$('.sidebar').classList.toggle('open'));
  document.addEventListener('click',e=>{
    if(e.target.closest('[data-close-modal]')) closeModal();
    const nav=e.target.closest('[data-page]'); if(nav){ navigate(nav.dataset.page); $('.sidebar').classList.remove('open'); }
    const openPlan=e.target.closest('[data-open-plan-global]'); if(openPlan) openPlanFromAny(openPlan.dataset.openPlanGlobal);
  });
}

async function enterApp(){
  if(!session) return;
  const {data,error}=await supabase.from('profiles').select('*').eq('id',session.user.id).maybeSingle();
  if(error||!data){ toast(error?.message||'Profilo non trovato.','error'); return; }
  profile=data; showApp();
  $('#userName').textContent=data.full_name||'Utente'; $('#userEmail').textContent=data.email||session.user.email||'';
  $('#userInitials').textContent=(data.full_name||session.user.email||'U').split(/\s+|@/).map(x=>x[0]).join('').slice(0,2).toUpperCase();
  $('#roleLabel').textContent=data.role==='nutritionist'?'Gestionale nutrizionale':'Area paziente';
  if(data.role==='nutritionist'){
    await loadPatients(); renderNav(nutritionistNav); navigate(currentPage && nutritionistNav.some(x=>x[0]===currentPage)?currentPage:'dashboard');
  }else{
    const {data:p,error:pe}=await supabase.from('patients').select('*').eq('user_id',session.user.id).maybeSingle();
    if(pe){toast(pe.message,'error');return;} selfPatient=p; renderNav(patientNav); navigate(currentPage && patientNav.some(x=>x[0]===currentPage)?currentPage:'patientHome');
  }
}
function renderNav(items){
  $('#mainNav').innerHTML=items.map(([page,icon,label])=>`<button class="nav-item ${page===currentPage?'active':''}" data-page="${page}">${icon}<span>${label}</span></button>`).join('');
}
async function navigate(page){
  currentPage=page; const meta=pageMeta[page]||pageMeta.dashboard; $('#pageEyebrow').textContent=meta[0]; $('#pageTitle').textContent=meta[1]; $('#pageSubtitle').textContent=meta[2];
  $$('#mainNav .nav-item').forEach(x=>x.classList.toggle('active',x.dataset.page===page));
  const root=$('#pageRoot'); root.innerHTML='<div class="card empty-state">Caricamento…</div>';
  try{
    if(page==='dashboard') await renderDashboard();
    else if(page==='patients') await renderPatients();
    else if(page==='visits') await renderVisits();
    else if(page==='bia') await renderBia();
    else if(page==='plans') await renderPlans();
    else if(page==='generator') await renderGenerator();
    else if(page==='foods') await renderFoods();
    else if(page==='documents') await renderDocuments();
    else if(page==='patientHome') await renderPatientHome();
    else if(page==='patientPlans') await renderPatientPlans();
    else if(page==='patientBia') await renderPatientBia();
    else if(page==='patientDocuments') await renderPatientDocuments();
  }catch(e){root.innerHTML=`<div class="card empty-state">${esc(e.message||String(e))}</div>`; toast(e.message||String(e),'error');}
}

async function renderDashboard(){

  const [patientCount,visitCount,planCount,biaCount,recentVisits,recentPlans]=await Promise.all([
    supabase.from('patients').select('id',{count:'exact',head:true}),
    supabase.from('visits').select('id',{count:'exact',head:true}),
    supabase.from('diet_plans').select('id',{count:'exact',head:true}),
    supabase.from('bia_records').select('id',{count:'exact',head:true}),
    supabase.from('visits').select('*').order('visit_date',{ascending:false}).limit(6),
    supabase.from('diet_plans').select('*').order('created_at',{ascending:false}).limit(6)
  ]);

function kpi(label,value,sub){const display=typeof value==='number'?fmt(value):esc(value??'—');return `<div class="card kpi-card"><span>${label}</span><strong>${display}</strong><small>${esc(sub||'')}</small></div>`;}
function visitTable(rows,limit=null){
  const list=limit?rows.slice(0,limit):rows; if(!list.length)return '<div class="empty-state">Nessuna visita.</div>';
  return `<div class="table-wrap"><table class="data-table"><thead><tr><th>Data</th><th>Paziente</th><th>Peso</th><th>Obiettivo</th><th>Prossima</th></tr></thead><tbody>${list.map(v=>`<tr><td>${dateIt(v.visit_date)}</td><td>${esc(patientName(v.patient_id))}</td><td>${v.weight_kg?fmt(v.weight_kg,1)+' kg':'—'}</td><td>${esc(v.goal||'—')}</td><td>${dateIt(v.next_visit)}</td></tr>`).join('')}</tbody></table></div>`;
}
function planMiniList(rows){if(!rows.length)return '<div class="empty-state">Nessun piano.</div>';return `<div class="history-list">${rows.map(p=>`<div class="history-row"><div><strong>${esc(p.title)}</strong><small>${esc(patientName(p.patient_id))} · ${dateTimeIt(p.created_at)}</small></div><span class="status-badge ${p.status}">${esc(p.status)}</span><button class="btn small" data-open-plan-global="${p.id}">Apri</button></div>`).join('')}</div>`;}
